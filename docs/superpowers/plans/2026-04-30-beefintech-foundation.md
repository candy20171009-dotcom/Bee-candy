**Goal:** Build a runnable Phase-1 BeeFintech web platform in the current repo with layered architecture, homepage, demo request flow, and dividend query flow using local mock data.

**Architecture:** Use Next.js App Router with clear boundaries across `app`/`components` (presentation), `features` (use-cases), `domains` (validation and types), and `repositories` (data access contracts + mock implementations). Route Handlers only parse HTTP and call use-cases. Data source is injected through a repository factory so Supabase can replace mock storage in Phase 2 without rewriting business logic.

**Tech Stack:** Next.js 16 (App Router), React 19, TypeScript, Tailwind CSS 4, ESLint, local JSON mock data.

```ts
// domains/demo/validation.ts
import type { DemoRequestInput } from "./types";

export type FieldError = { field: string; message: string };

export function validateDemoRequestInput(input: DemoRequestInput): FieldError[] {
  const errors: FieldError[] = [];
  if (!input.name.trim()) errors.push({ field: "name", message: "姓名不能为空" });
  if (!input.company.trim()) errors.push({ field: "company", message: "公司名称不能为空" });
  if (!input.role.trim()) errors.push({ field: "role", message: "用户角色不能为空" });
  if (!input.interestedService.trim()) errors.push({ field: "interestedService", message: "请选择感兴趣服务" });

  const hasContact = input.contact.trim().length > 0;
  const hasEmail = input.email.trim().length > 0;
  if (!hasContact && !hasEmail) {
    errors.push({ field: "contact", message: "联系方式或邮箱至少填写一项" });
    errors.push({ field: "email", message: "联系方式或邮箱至少填写一项" });
  }
  return errors;
}
```

```ts
import type { DividendQuery, DividendRecord } from "@/domains/dividend/types";
import type { DividendRepository } from "@/repositories/contracts/dividend-repository";

    return (data as DividendRecord[]).filter((row) => {

      if (query.productName && row.productName !== query.productName) return false;

      if (query.year && row.year !== query.year) return false;
      return true;
    });
  }
```

```ts
// lib/http/api-response.ts
import { NextResponse } from "next/server";
import { ApiError } from "./api-error";

export function ok<T>(data: T, status = 200) {
  return NextResponse.json({ success: true, data }, { status });
}

export function fail(error: unknown) {
  if (error instanceof ApiError) {
    return NextResponse.json(
      { success: false, error: { code: error.code, message: error.message, details: error.details } },
      { status: error.status },
    );
  }
  return NextResponse.json(
    { success: false, error: { code: "INTERNAL_ERROR", message: "服务暂时不可用，请稍后再试" } },
    { status: 500 },
  );
}
```

```ts
import { fail, ok } from "@/lib/http/api-response";
import { getRepositories } from "@/repositories/repository-factory";
import { submitDemoRequest } from "@/features/demo/submit-demo-request";

export async function POST(request: Request) {
  try {
    const payload = await request.json();
    const { demoRequestRepository } = getRepositories();
    const saved = await submitDemoRequest(demoRequestRepository, payload);
    return ok({ requestId: saved.id, createdAt: saved.createdAt });
  } catch (error) {
    return fail(error);
  }
}
```

```ts
// app/api/dividend/route.ts
import { fail, ok } from "@/lib/http/api-response";
import { getRepositories } from "@/repositories/repository-factory";
import { searchDividendRates } from "@/features/dividend/search-dividend-rates";

export async function GET(request: Request) {
  try {
    const { searchParams } = new URL(request.url);
    const yearValue = searchParams.get("year");
    const query = {
      insuranceCompany: searchParams.get("insuranceCompany") ?? undefined,
      productName: searchParams.get("productName") ?? undefined,
      currency: searchParams.get("currency") ?? undefined,
      year: yearValue ? Number(yearValue) : undefined,
    };
    const { dividendRepository } = getRepositories();
    const items = await searchDividendRates(dividendRepository, query);
    return ok({ items, total: items.length });
  } catch (error) {
    return fail(error);
  }
}
```

```tsx
// components/demo/demo-form.tsx
"use client";

import { useState } from "react";

const initial = { name: "", company: "", contact: "", email: "", role: "", interestedService: "", message: "" };

export function DemoForm() {
  const [form, setForm] = useState(initial);
  const [status, setStatus] = useState<"idle" | "loading" | "success" | "error">("idle");
  const [message, setMessage] = useState("");

  function update<K extends keyof typeof initial>(key: K, value: string) {
    setForm((prev) => ({ ...prev, [key]: value }));
  }

  async function onSubmit(e: React.FormEvent) {
    e.preventDefault();
    setStatus("loading");
    setMessage("");
    const res = await fetch("/api/demo-request", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(form),
    });
    const json = await res.json();
    if (json.success) {
      setStatus("success");
      setMessage("预约提交成功，我们会尽快联系你。");
      setForm(initial);
      return;
    }
    setStatus("error");
    setMessage(json.error?.message || "提交失败，请稍后再试。");
  }

  return (
    <form onSubmit={onSubmit} className="space-y-4 rounded-xl border p-6">
      <input className="w-full rounded border p-2" placeholder="姓名" value={form.name} onChange={(e) => update("name", e.target.value)} />
      <input className="w-full rounded border p-2" placeholder="公司名称" value={form.company} onChange={(e) => update("company", e.target.value)} />
      <input className="w-full rounded border p-2" placeholder="联系方式" value={form.contact} onChange={(e) => update("contact", e.target.value)} />
```

```tsx
"use client";

import { useState } from "react";
import type { DividendRecord } from "@/domains/dividend/types";
import { DividendFilter } from "@/components/dividend/dividend-filter";
import { DividendResults } from "@/components/dividend/dividend-results";

export default function DividendToolPage() {
  const [items, setItems] = useState<DividendRecord[]>([]);
  const [hasSearched, setHasSearched] = useState(false);
  const [error, setError] = useState("");

  async function onSearch(query: Record<string, string | undefined>) {
    setHasSearched(true);
    setError("");
    const qs = new URLSearchParams();
    Object.entries(query).forEach(([k, v]) => {
      if (v?.trim()) qs.set(k, v);
    });
    const res = await fetch(`/api/dividend?${qs.toString()}`);
    const json = await res.json();
    if (!json.success) {
      setItems([]);
      setError(json.error?.message || "数据暂时无法加载，请稍后再试");
      return;
    }
    setItems(json.data.items);
  }

  return (
    <main className="mx-auto max-w-6xl space-y-6 p-6">
      <h1 className="text-2xl font-semibold">分红达成率查询工具</h1>
      <DividendFilter onSearch={onSearch} />
      {error ? <p className="text-red-600">{error}</p> : null}
      <DividendResults items={items} hasSearched={hasSearched} />
    </main>
  );
}
```
